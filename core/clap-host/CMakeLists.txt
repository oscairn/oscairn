# The CLAP host is Rust (rust/, built into the engine core's static library by core/CMakeLists.txt, which also builds
# kick_clap_host, its C++ face for C++ callers, from plugin-host/src/kick_clap.cpp). Here: the one C++ file the Rust host
# calls, the host's own windows for plugins' editors (Cocoa on macOS, none elsewhere yet), and the test fixtures.
add_library(oscairn_clap_windows STATIC src/WindowBridge.cpp)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(oscairn_clap_windows PRIVATE src/Window.mm)
    set_source_files_properties(src/Window.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(oscairn_clap_windows PUBLIC "-framework AppKit")
else()
    target_sources(oscairn_clap_windows PRIVATE src/Window.cpp)
endif()
target_compile_features(oscairn_clap_windows PUBLIC cxx_std_17)
set_target_properties(oscairn_clap_windows PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(oscairn_clap_windows PUBLIC include PRIVATE ../plugin-host/include third_party/clap/include ../clap-ext/include) # Host.h's runWindowEvents
if(MSVC)
    target_compile_options(oscairn_clap_windows PRIVATE /W4)
else()
    target_compile_options(oscairn_clap_windows PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(kick_clap_sanitizers INTERFACE)
option(KICK_CLAP_SANITIZE "Enable address and undefined-behavior sanitizers for the CLAP fixtures" OFF)
if(KICK_CLAP_SANITIZE AND NOT MSVC)
    target_compile_options(kick_clap_sanitizers INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kick_clap_sanitizers INTERFACE -fsanitize=address,undefined)
endif()
enable_testing()
add_subdirectory(tests)
